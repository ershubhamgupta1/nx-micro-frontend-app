npx create-nx-workspace@latest nx-mfe-poc \
  --workspaceType=integrated \
  --preset=apps \
  --nxCloud=skip \
  --interactive=false



  cd nx-mfe-poc

npm install -D @nx/react @nx/webpack @nx/jest

nx g @nx/react:host \
  --name=shell \
  --directory=apps/shell \
  --bundler=webpack \
  --style=css \
  --unitTestRunner=jest \
  --e2eTestRunner=none

nx g @nx/react:remote \
  --name=dispatch_messages \
  --host=shell \
  --directory=apps/dispatch_messages \
  --bundler=webpack \
  --style=css \
  --unitTestRunner=jest \
  --e2eTestRunner=none
  
nx reset 

nx serve shell (this will automatically load both shell and dispatch_messages)


================================

1) Create build for shell and dispatch_messages:  nx build shell 
  Desc: it will automatically build both shell and dispatch_messages

2) How to validate the build 

  npm install -g serve
  cd dist/apps/shell
  serve

3) Now open the url given by serve in your browser and you should see the shell app with dispatch_messages loaded
Note: see you can see the app.tsx in sources tab, you can see the files are not hashed, only one separate file for the vendors
